#  機率密度圖
import seaborn as sns
import matplotlib.pyplot as plt

# ========== 欄位名稱（中文化） ==========
col_map = {
    'strategy_power': '壓制',
    'strategy_emotion': '情緒',
    'strategy_blame': '指責',
    'score_fear': '恐懼',
    'score_obligation': '責任',
    'score_guilt': '罪惡',
    'score_overall': '總分'
}

numeric_cols = list(col_map.keys())

# ========== KDE 圖 ==========
sns.set(style="whitegrid", font_scale=1.05)

fig, axes = plt.subplots(nrows=2, ncols=4, figsize=(14, 8))
axes = axes.flatten()

for i, col in enumerate(numeric_cols):
    ax = axes[i]

    sns.kdeplot(
        df_main[col],
        label="主文（level=1）",
        color="blue",
        fill=True,
        alpha=0.35,
        ax=ax
    )
    sns.kdeplot(
        df_reply[col],
        label="留言（level=2）",
        color="orange",
        fill=True,
        alpha=0.35,
        ax=ax
    )

    ax.set_title(f"分布比較：{col_map[col]}", fontproperties=winfont02)
    ax.set_xlabel(col_map[col], fontproperties=winfont02)
    ax.set_ylabel("密度（Density）", fontproperties=winfont02)

    # Legend（使用中文字型）
    ax.legend(prop=winfont02, loc="upper right")

# 移除多餘的 subplot
fig.delaxes(axes[-1])

plt.tight_layout()
plt.show()

主文分布高峰更明顯
留言大量集中在 0、1
主文的跨度較大（從 0–5 都有）

# 因為不可能是常態
from scipy.stats import ks_2samp
print("Kolmogorov–Smirnov Test（主文 vs 留言）")
print("-" * 60)

results = []

for col in numeric_cols:
    data_main = df_main[col].dropna()
    data_reply = df_reply[col].dropna()

    ks_stat, p_value = ks_2samp(data_main, data_reply)

    results.append((col, ks_stat, p_value))

    print(f"{col:18s} | KS stat = {ks_stat:.4f} | p = {p_value:.4e} | "
          f"{'**顯著差異**' if p_value < 0.05 else '不顯著'}")
# h0：主文&主文+留言 來自相同分布
# h1：主文&主文+留言 來自不同分布
