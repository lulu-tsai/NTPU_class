import sqlite3
import pandas as pd

DB_FILE = "D:/NTPU_class/paper/code/mobile01_clawler/ntpu_paper.sqlite"

conn = sqlite3.connect(DB_FILE)

df_eda = pd.read_sql_query("""
    -- A) 留言評分：comment_id 有值 → 每個 comment 取最新
    SELECT 
        e.comment_id,
        e.level,
        e.created_at AS eval_created_at,
        e.strategy_power,
        e.strategy_emotion,
        e.strategy_blame,
        e.score_fear,
        e.score_obligation,
        e.score_guilt,
        e.score_overall,
        e.main_strategy,
        e.main_strategy_detail,
        e.pua_source,
        e.confidence,
        e.model_name,
        e.model_version,
        e.knowledge_base,
        e.article_id,
        e.link_id
    FROM eb_evaluation e
    JOIN (
        SELECT comment_id, MAX(created_at) AS max_created_at
        FROM eb_evaluation
        WHERE comment_id IS NOT NULL
        GROUP BY comment_id
    ) latest_c
      ON e.comment_id = latest_c.comment_id
     AND e.created_at = latest_c.max_created_at

    UNION ALL

    -- B) 主文評分：comment_id 為空 → 每篇 article 取最新
    SELECT 
        e.comment_id,
        e.level,
        e.created_at AS eval_created_at,
        e.strategy_power,
        e.strategy_emotion,
        e.strategy_blame,
        e.score_fear,
        e.score_obligation,
        e.score_guilt,
        e.score_overall,
        e.main_strategy,
        e.main_strategy_detail,
        e.pua_source,
        e.confidence,
        e.model_name,
        e.model_version,
        e.knowledge_base,
        e.article_id,
        e.link_id
    FROM eb_evaluation e
    JOIN (
        SELECT article_id, MAX(created_at) AS max_created_at
        FROM eb_evaluation
        WHERE comment_id IS NULL
        GROUP BY article_id
    ) latest_a
      ON e.article_id = latest_a.article_id
     AND e.created_at = latest_a.max_created_at
    WHERE e.comment_id IS NULL
""", conn)


conn.close()

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from pathlib import Path

col_map = {
    'strategy_power': '壓制',
    'strategy_emotion': '情緒',
    'strategy_blame': '指責',
    'score_fear': '恐懼',
    'score_obligation': '責任',
    'score_guilt': '罪惡',
    'score_overall': '總分'
}
score_order = [0, 1, 2, 3, 4, 5]

sns.set(style="whitegrid", font_scale=1.05)

def _coerce_scores_0_5(df: pd.DataFrame) -> pd.DataFrame:
    """把分數欄位轉成可畫的 0–5 整數（會把不合法值變 NaN）。"""
    out = df.copy()
    for c in col_map.keys():
        # 轉數字（字串也能轉），不能轉的變 NaN
        out[c] = pd.to_numeric(out[c], errors="coerce")

        # 如果你分數本來就是整數，這行不影響；若是 0.0/1.0 也會變整數
        out[c] = out[c].round()

        # 只保留 0–5，其它丟掉避免干擾
        out.loc[~out[c].between(0, 5), c] = pd.NA

        # 轉成 pandas 的可空整數
        out[c] = out[c].astype("Int64")
    return out

def plot_and_save_big(df: pd.DataFrame, title_text: str, out_path: str | None = None):
    df = _coerce_scores_0_5(df)

    fig, axes = plt.subplots(2, 4, figsize=(18, 8))
    axes = axes.flatten()

    for i, (col, cname) in enumerate(col_map.items()):
        ax = axes[i]
        sns.countplot(
            data=df,
            x=col,
            order=score_order,
            ax=ax
        )
        ax.set_title(cname, fontproperties=winfont02)
        ax.set_xlabel("分數（0–5）", fontproperties=winfont02)
        ax.set_ylabel("次數（Count）", fontproperties=winfont02)
        ax.set_xticks(score_order)

        # 顯示該構面的有效樣本數（避免你看到空但其實是全 NaN）
        n_valid = df[col].notna().sum()
        ax.text(0.98, 0.95, f"n={n_valid}", transform=ax.transAxes,
                ha="right", va="top", fontproperties=winfont02)

    # 移除多餘 subplot
    fig.delaxes(axes[-1])

    fig.suptitle(title_text, fontproperties=winfont02, fontsize=18, y=1.02)
    plt.tight_layout()

    if out_path:
        out_path = Path(out_path)
        out_path.parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(out_path, dpi=300, bbox_inches="tight")

    plt.show()
    plt.close(fig)

# 主文一張（可存檔）
plot_and_save_big(df_main, "主文（level=1）各構面分數次數分布（0–5）", "main_level1_count_grid.png")

# ===== 1. 整理資料（確保型態） =====
def prep_df(df):
    out = df[["score_overall", "confidence"]].copy()
    out["score_overall"] = pd.to_numeric(out["score_overall"], errors="coerce")
    out["confidence"] = pd.to_numeric(out["confidence"], errors="coerce")
    return out.dropna(subset=["score_overall", "confidence"])

主文（level=1）

  • 點多集中在 confidence 約 0.8–1.0
  • 不論 overall 分數高低，模型多半給出 中高以上信心
  • 趨勢線呈現 輕微下降
  • 隨 overall 分數提高，confidence 的平均趨勢略為下降，同時其分布範圍呈現較大離散程度

留言（level=2）

  • 高 overall（3–5）時，confidence 明顯集中在 0.9 以上
  • 低 overall（0–1）時，confidence 分布較為分散（甚至有偏低的點）
  • 趨勢線同樣呈現 輕微下降
